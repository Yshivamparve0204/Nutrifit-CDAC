🟢 STEP 3: INSERT ALL POSSIBLE DIET PLANS

(Goal × Condition × FoodPreference × WeightCategory)

👉 3 × 6 × 3 × 3 = 162 diet plans

INSERT INTO DietPlans (Goal, ConditionId, FoodPreference, WeightCategory)
SELECT g.Goal, h.HealthConditionId, f.FoodPreference, w.WeightCategory
FROM (VALUES 
('Weight Loss'),
('Muscle Gain'),
('Fitness')
) g(Goal)


CROSS JOIN HealthConditions h


CROSS JOIN (VALUES 
('Vegan'),
('Vegetarian'),
('Non-Vegetarian')
) f(FoodPreference)


CROSS JOIN (VALUES 
('Underweight'),
('Normal'),
('Overweight')
) w(WeightCategory);

🟢 STEP 4: CONNECT EVERY PLAN WITH 5 DAILY MEALS

(breakfast, snack, lunch, snack2, dinner)

;WITH FoodPool AS
(
    SELECT 
        dp.DietId,
        f.FoodId,
        ROW_NUMBER() OVER (PARTITION BY dp.DietId ORDER BY NEWID()) AS rn
    FROM DietPlans dp
    JOIN Foods f ON f.FoodType = dp.FoodPreference
)


INSERT INTO DietPlanFoods (DietId, FoodId, MealType)
SELECT 
    DietId,
    FoodId,
    CASE rn
        WHEN 1 THEN 'breakfast'
        WHEN 2 THEN 'snack'
        WHEN 3 THEN 'lunch'
        WHEN 4 THEN 'snack2'
        WHEN 5 THEN 'dinner'
    END
FROM FoodPool
WHERE rn <= 5;
